#ifndef _AP3216C_H_

enum ap3216c_mode
{
    AP3216C_MODE_POWER_DOWN,      // Power down (Default)
    AP3216C_MODE_ALS,             // ALS function active
    AP3216C_MODE_PS,              // PS+IR function active
    AP3216C_MODE_ALS_AND_PS,      // ALS and PS+IR functions active
    AP3216C_MODE_SW_RESET,        // SW reset
    AP3216C_MODE_ALS_ONCE,        // ALS function once
    AP3216C_MODE_PS_ONCE,         // PS+IR function once
    AP3216C_MODE_ALS_AND_PS_ONCE, // ALS and PS+IR functions once
};

enum ap3216c_int_clear
{
    AP3216C_INT_AUTO_CLR,  // automatic cleared by reading data registers
    AP3216C_INT_WRITE_CLR, // cleared by write 1
};

#define AP3216C_DRV_NAME "ap3216c"
/* AP3216C system reg address*/
#define AP3216C_CONF_REG 0x00
#define AP3216C_INT_REG 0x01
#define AP3216C_INT_CLR_REG 0x02
#define AP3216C_IR_LDAT_REG 0x0A
#define AP3216C_IR_HDAT_REG 0x0B
#define AP3216C_ALS_LDAT_REG 0x0C
#define AP3216C_ALS_HDAT_REG 0x0D
#define AP3216C_PS_LDAT_REG 0x0E
#define AP3216C_PS_HDAT_REG 0x0F

enum ap3216c_als_gain
{
    AP3216C_ALS_GAIN_035,
    AP3216C_ALS_GAIN_00788,
    AP3216C_ALS_GAIN_00197,
    AP3216C_ALS_GAIN_00049,
};

/* AP3216C ALS reg address*/
#define AP3216C_ALS_CONF_REG 0x10
#define AP3216C_ALS_CAL_REG 0x19
#define AP3216C_ALS_LTH1_REG 0x1A // low threshold
#define AP3216C_ALS_LTH2_REG 0x1B
#define AP3216C_ALS_HTH1_REG 0x1C // high threshold
#define AP3216C_ALS_HTH2_REG 0x1D

/* AP3216C PS reg address*/
#define AP3216C_PS_CONF_REG 0x20
#define AP3216C_PS_LED_REG 0x21
#define AP3216C_PS_INT_REG 0x22
#define AP3216C_PS_AVTM_REG 0x23    // average time
#define AP3216C_PS_LED_WTM_REG 0x24 // led wait time
#define AP3216C_PS_LCAL_REG 0x28
#define AP3216C_PS_HCAL_REG 0x29
#define AP3216C_PS_LTH1_REG 0x2A
#define AP3216C_PS_LTH2_REG 0x2B
#define AP3216C_PS_HTH1_REG 0x2C
#define AP3216C_PS_HTH2_REG 0x2D
#define AP3216C_MAX_REG 0x2E

/* AP3216C reg masks */
#define AP3216C_CONF_MASK 0x07
#define AP3216C_IR_VALID_MASK 0x80
#define AP3216C_PS_VALID_MASK 0x40
#define AP3216C_OBJ_DETECT_MASK 0x80
#define AP3216C_IR_LDAT_MASK 0x03
#define AP3216C_IR_HDAT_MASK 0xFF
#define AP3216C_ALS_LDAT_MASK 0xFF
#define AP3216C_ALS_HDAT_MASK 0xFF
#define AP3216C_PS_LDAT_MASK 0x0F
#define AP3216C_PS_HDAT_MASK 0x3F

/* AP3216C reg bits */
#define AP3216C_IR_LDAT_BITS 2
#define AP3216C_IR_HDAT_BITS 8
#define AP3216C_ALS_LDAT_BITS 8
#define AP3216C_ALS_HDAT_BITS 8
#define AP3216C_PS_LDAT_BITS 4
#define AP3216C_PS_HDAT_BITS 6

/* AP3216C delay ms */
#define AP3216C_RST_DELAY_MS 10

/**
 * AP3216C interrupt request status
 * @AP3216C_IRQ_ALS: Als interrupt is triggered when the adc reading remains
 * outside the threshold range for longer than the configured persistence time.
 * @AP3216C_IRQ_PS: Ps interrupt is triggered when the adc reading remains
 * outside the threshold range for longer than the configured persistence time.
 * @AP3216C_IRQ_ALS_AND_PS:
 */
#define AP3216C_IRQ_ALS (1 << 0)
#define AP3216C_IRQ_PS (1 << 1)
#define AP3216C_IRQ_ALS_AND_PS (AP3216C_IRQ_ALS | AP3216C_IRQ_PS)

#endif

struct ap3216c_data
{
    struct i2c_client *client;
    struct regmap *regmap;
    struct mutex lock;

    int irq;
};

static int ap3216c_read_ir_data(struct ap3216c_data *data, int *val)
{
    int ret;
    uint8_t ir[2];

    ret = regmap_bulk_read(data->regmap, AP3216C_IR_LDAT_REG, ir, 2);
    if (ret)
        return -EINVAL;

    if (ir[0] & AP3216C_IR_VALID_MASK)
    {
        pr_debug("invalid data: ap3216c ir\n");
        return -EINVAL;
    }

    *val = ((ir[1] & AP3216C_IR_HDAT_MASK) << AP3216C_IR_LDAT_BITS) | (ir[0] & AP3216C_IR_LDAT_MASK);

    return IIO_VAL_INT;
}

static int ap3216c_read_als_data(struct ap3216c_data *data, int *val)
{
    int ret;
    uint8_t als[2];

    ret = regmap_bulk_read(data->regmap, AP3216C_ALS_LDAT_REG, als, 2);
    if (ret)
        return -EINVAL;

    *val = ((als[1] & AP3216C_ALS_HDAT_MASK) << AP3216C_ALS_LDAT_BITS) | (als[0] & AP3216C_ALS_LDAT_MASK);

    return IIO_VAL_INT;
}

static int ap3216c_read_ps_data(struct ap3216c_data *data, int *val)
{
    int ret;
    uint8_t ps[2];

    ret = regmap_bulk_read(data->regmap, AP3216C_PS_LDAT_REG, ps, 2);
    if (ret)
        return -EINVAL;

    if (ps[0] & AP3216C_PS_VALID_MASK)
    {
        pr_debug("invalid data: ap3216c ps\n");
        return -EINVAL;
    }

    *val = ((ps[1] & AP3216C_PS_HDAT_MASK) << AP3216C_PS_LDAT_BITS) | (ps[0] & AP3216C_PS_LDAT_MASK);

    return IIO_VAL_INT;
}

static int ap3216c_read_raw(struct iio_dev *indio_dev,
                            struct iio_chan_spec const *chan,
                            int *val, int *val2, long mask)
{
    struct ap3216c_data *data = iio_priv(indio_dev);
    int ret;
    switch (mask)
    {
    case IIO_CHAN_INFO_RAW:
        mutex_lock(&data->lock);
        switch (chan->type)
        {
        case IIO_LIGHT:
            ret = ap3216c_read_als_data(data, val);
            break;
        case IIO_PROXIMITY:
            ret = ap3216c_read_ps_data(data, val);
            break;
        case IIO_INTENSITY:
            ret = ap3216c_read_ir_data(data, val);
            break;
        default:
            ret = -EINVAL;
            break;
        }
        mutex_unlock(&data->lock);
        return ret;
        break;
    case IIO_CHAN_INFO_SCALE:
        switch (chan->type)
        {
        case IIO_LIGHT:
        case IIO_PROXIMITY:
        case IIO_INTENSITY:
            return IIO_VAL_INT;
            break;
        default:
            break;
        }
        break;
    default:
        break;
    }
    return -EINVAL;
}

static int ap3216c_read_als_thresh(struct ap3216c_data *data, int *thresh_low, int *thresh_high)
{
    int ret;
    uint8_t buff[4];

    ret = regmap_bulk_read(data->regmap, AP3216C_ALS_LTH1_REG, buff, 4);
    if (ret)
        return -EINVAL;

    *thresh_low = buff[1] << 8 | buff[0];
    *thresh_high = buff[3] << 8 | buff[2];

    return 0;
}

static int ap3216c_read_ps_thresh(struct ap3216c_data *data, int *thresh_low, int *thresh_high)
{
    int ret;
    uint8_t buff[4];

    ret = regmap_bulk_read(data->regmap, AP3216C_PS_LTH1_REG, buff, 4);
    if (ret)
        return -EINVAL;

    *thresh_low = buff[1] << 8 | buff[0];
    *thresh_high = buff[3] << 8 | buff[2];

    return 0;
}

static int ap3216c_read_event_value(struct iio_dev *indio_dev,
                                    const struct iio_chan_spec *chan,
                                    enum iio_event_type type,
                                    enum iio_event_direction dir,
                                    enum iio_event_info info, int *val, int *val2)
{
    struct ap3216c_data *data = iio_priv(indio_dev);
    int ret;

    uint8_t buff[2];

    if (dir == IIO_EV_DIR_FALLING)
    {
        ret = regmap_bulk_read(data->regmap, AP3216C_ALS_LTH1_REG, buff, 2);
        if (ret)
            return -EINVAL;

        *val = buff[1] << 8 | buff[0];
    }

    if (dir == IIO_EV_DIR_RISING)
    {
        ret = regmap_bulk_read(data->regmap, AP3216C_ALS_HTH1_REG, buff, 2);
        if (ret)
            return -EINVAL;

        *val = buff[1] << 8 | buff[0];
    }

    return IIO_VAL_INT;
}

static int ap3216c_write_event_value(struct iio_dev *indio_dev,
                                     const struct iio_chan_spec *chan,
                                     enum iio_event_type type,
                                     enum iio_event_direction dir,
                                     enum iio_event_info info, int val, int val2)
{
    struct ap3216c_data *data = iio_priv(indio_dev);
    int ret;
    uint8_t buf[2];
    buf[0] = val & 0xff;
    buf[1] = val >> 8;
    if (dir == IIO_EV_DIR_FALLING)
    {
        ret = regmap_bulk_write(data->regmap, AP3216C_ALS_LTH1_REG, buf, 2);
    }

    if (dir == IIO_EV_DIR_RISING)
    {
        ret = regmap_bulk_write(data->regmap, AP3216C_ALS_HTH1_REG, buf, 2);
    }

    if (ret)
        return -EINVAL;

    return IIO_VAL_INT;
}

static bool ap3216c_is_volatile_reg(struct device *dev, unsigned int reg)
{
    switch (reg)
    {
    case AP3216C_IR_LDAT_REG:
    case AP3216C_IR_HDAT_REG:
    case AP3216C_ALS_LDAT_REG:
    case AP3216C_ALS_HDAT_REG:
    case AP3216C_PS_LDAT_REG:
    case AP3216C_PS_HDAT_REG:
        return true;
    default:
        return false;
    }
}

static int ap3216c_hw_init(struct ap3216c_data *data)
{
    int ret;

    ret = regmap_write(data->regmap, AP3216C_CONF_REG, AP3216C_MODE_SW_RESET);
    if (ret < 0)
        return ret;

    msleep(AP3216C_RST_DELAY_MS);

    // ret = regmap_update_bits(data->regmap, AP3216C_ALS_CONF_REG, 0x0f, AP3216C_ALS_GAIN_035);
    // if (ret < 0)
    //     return ret;

    ret = regmap_write(data->regmap, AP3216C_CONF_REG, AP3216C_MODE_ALS_AND_PS);
    if (ret < 0)
        return ret;

    ret = regmap_write(data->regmap, AP3216C_INT_CLR_REG, AP3216C_INT_AUTO_CLR);
    if (ret < 0)
        return ret;

    return 0;
}

static int ap3216c_power_off(struct ap3216c_data *data)
{
    int ret, val;

    mutex_lock(&data->lock);
    ret = regmap_write(data->regmap, AP3216C_CONF_REG, AP3216C_MODE_POWER_DOWN);
    if (ret < 0)
        return ret;

    ret = regmap_read(data->regmap, AP3216C_CONF_REG, &val);
    if (ret < 0)
        return ret;

    if (val != AP3216C_MODE_POWER_DOWN)
    {
        printk("AP3216C failed to write mode: %d\n", AP3216C_MODE_POWER_DOWN);
        return -EINVAL;
    }
    mutex_unlock(&data->lock);
    return 0;
}

static irqreturn_t ap3216c_irq_handler(int irq, void *dev_id)
{
    struct iio_dev *indio_dev = i2c_get_clientdata(dev_id);
    struct ap3216c_data *data = iio_priv(indio_dev);
    int irq_status, ret;
    int adc_val = 0;
    int thresh_high = 0, thresh_low = 0;

    ret = regmap_read(data->regmap, AP3216C_INT_REG, &irq_status);
    if (ret < 0)
        return ret;

    switch (irq_status)
    {
    case AP3216C_IRQ_ALS:
        ap3216c_read_als_data(data, &adc_val);
        ap3216c_read_als_thresh(data, &thresh_low, &thresh_high);
        if (adc_val > thresh_high)
            iio_push_event(indio_dev,
                           IIO_UNMOD_EVENT_CODE(IIO_LIGHT,
                                                0,
                                                IIO_EV_TYPE_THRESH,
                                                IIO_EV_DIR_RISING),
                           iio_get_time_ns());
        else if (adc_val < thresh_low)
            iio_push_event(indio_dev,
                           IIO_UNMOD_EVENT_CODE(IIO_LIGHT,
                                                0,
                                                IIO_EV_TYPE_THRESH,
                                                IIO_EV_DIR_FALLING),
                           iio_get_time_ns());
        break;
    // case AP3216C_IRQ_PS:
    //     iio_push_event(indio_dev,
    //                    IIO_UNMOD_EVENT_CODE(IIO_PROXIMITY,
    //                                         0,
    //                                         IIO_EV_TYPE_THRESH,
    //                                         IIO_EV_DIR_NONE),
    //                    iio_get_time_ns());
    //     break;
    // case AP3216C_IRQ_ALS_AND_PS:
    //     iio_push_event(indio_dev,
    //                    IIO_UNMOD_EVENT_CODE(IIO_LIGHT,
    //                                         0,
    //                                         IIO_EV_TYPE_THRESH,
    //                                         IIO_EV_DIR_NONE),
    //                    iio_get_time_ns());
    //     iio_push_event(indio_dev,
    //                    IIO_UNMOD_EVENT_CODE(IIO_PROXIMITY,
    //                                         0,
    //                                         IIO_EV_TYPE_THRESH,
    //                                         IIO_EV_DIR_NONE),
    //                    iio_get_time_ns());
    //     break;
    default:
        break;
    }

    return IRQ_HANDLED;
}
int ap3216c_probe(struct i2c_client *client, const struct i2c_device_id *id)
{
    struct ap3216c_data *data;
    struct iio_dev *indio_dev;
    int ret;

    dev_info(&client->dev, "ap3216c probe enter\n");

    /* */
    indio_dev = devm_iio_device_alloc(&client->dev, sizeof(*data));
    if (!indio_dev)
        return -ENOMEM;

    data = iio_priv(indio_dev);
    i2c_set_clientdata(client, indio_dev);
    data->client = client;

    mutex_init(&data->lock);

    ret = devm_request_irq(&client->dev,
                           client->irq,
                           ap3216c_irq_handler,
                           IRQF_TRIGGER_RISING,
                           "ap3216c_irq",
                           client);
    if (ret)
    {
        dev_info(&client->dev, "request irq failed: %d\n", ret);
        return ret;
    }

    /* */
    data->regmap = devm_regmap_init_i2c(client, &ap3216c_regmap_config);
    if (IS_ERR(data->regmap))
    {
        dev_info(&client->dev, "ap3216c regmap init failed\n");
        return PTR_ERR(data->regmap);
    }

    indio_dev->info = &ap3216c_info;
    indio_dev->name = AP3216C_DRV_NAME;
    indio_dev->modes = INDIO_DIRECT_MODE;
    indio_dev->channels = ap3216c_channels;
    indio_dev->num_channels = ARRAY_SIZE(ap3216c_channels);

    indio_dev->dev.parent = &client->dev;

    /* */
    ret = ap3216c_hw_init(data);
    if (ret)
        return ret;

    ret = devm_iio_device_register(&client->dev, indio_dev);

    dev_info(&client->dev, "ap3216c probe success\n");

    return ret;
}

int ap3216c_remove(struct i2c_client *client)
{
    struct iio_dev *indio_dev = i2c_get_clientdata(client);
    struct ap3216c_data *data = iio_priv(indio_dev);
    dev_info(&client->dev, "ap3216c remove\n");
    return ap3216c_power_off(data);
}
