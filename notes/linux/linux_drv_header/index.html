    raw_spin_lock_irqsave(spinlock_check(lock), flags); \
} while (0)

{
    raw_spin_unlock_irqrestore(&lock->rlock, flags);
}

linux/timer.h#

#include <linux/timer.h>
struct timer_list timer;
//旧版
//传参是unsigned long
init_timer(&dev->timer);

setup_timer(&dev->timer, timer_callback, (unsigned long)dev);

mod_timer(&dev->timer, jiffies + msecs_to_jiffies(20));
del_timer_sync(&dev->timer);

static void timer_callback(unsigned long data)
{
    struct dev_t *dev = (struct dev_t *) data;
}

//新版
//原本的传参被废弃，必须传struct timer_list *
//别的参数通过container_of / from_timer====>类似通过类成员反推类

timer_setup(&dev->timer, timer_callback, 0);

mod_timer(&dev->timer, jiffies + msecs_to_jiffies(20));

del_timer_sync(&dev->timer);

static void timer_callback(struct timer_list * timer)
{
    struct dev_t *dev = from_timer(dev, timer, timer);
}

linux/workqueue.h#

#include <linux/workqueue.h>

struct work_struct {
    atomic_long_t data;
    struct list_head entry;
    work_func_t func;
};

struct delayed_work {
    struct work_struct work;
    struct timer_list timer;

    struct workqueue_struct *wq;
    int cpu;
};

#define INIT_WORK(_work, _func) __INIT_WORK((_work), (_func), 0)
// _work: struct work_struct 
// _func: void (*)(struct work_struct *work)
#define __INIT_WORK(_work, _func, _onstack) \
    do {    \
        __init_work((_work), _onstack); \
        (_work)->data = (atomic_long_t) WORK_DATA_INIT(); \
        INIT_LIST_HEAD(&(_work)->entry); \
        (_work)->func = (_func); \
    } while (0)

#define INIT_DELAYED_WORK(_work, _func) __INIT_DELAYED_WORK(_work, _func, _tflags)

#define __INIT_DELAYED_WORK(_work, _func, _tflags) \
    do {    \
        INIT_WORK(&(_work)->work, (_func)); \
        __setup_timer(&(_work)->timer, delayed_work_timer_fn,   \
                    (unsigned long)(_work), \
                    (_tflags) | TIMER_IRQSAFE); \
    } while (0)

extern bool cancel_delayed_work_sync(struct delayed_work *dwork);

static inline bool schedule_delayed_work(struct delayed_work *dwork,
                    unsigned long delay)
{
    return queue_delayed_work(system_wq, dwork, delay);
}
