LED驱动实验#

刚创建myboard.dts。目标是添加一个GPIO节点来驱动LED。看着dts里边的内容，有点不知所措。

手边有imx6ull的数据手册。还有内核源码。看一看内核源码里的binding文档。

binding目录#

内核binding目录下查看gpio规范。/Documentation/devicetree/bindings/gpio/gpio.txt

仔细看会发现两个node 一个是pinctrl，一个是gpio。

pinctrl node and gpio node#

如果做过8位mcu、32位mcu的裸机，那就会对驱动很熟悉，第一步就是查手册配寄存器做初始化。

PINCTRL Subsystem#

Linux PINCTRL subsystem 链接点击👉pin-control.html

大概就是定义了有多少个引脚，编号是多少，名字叫啥。然后注册进linux, fsl官方把这块做好了。

static struct of_device_id imx6ul_pinctrl_of_match[] = {
	{ .compatible = "fsl,imx6ul-iomuxc", .data = &imx6ul_pinctrl_info, },
	{ .compatible = "fsl,imx6ull-iomuxc-snvs", .data = &imx6ull_snvs_pinctrl_info, },
	{ /* sentinel */ }
};

static int imx6ul_pinctrl_probe(struct platform_device *pdev)
{
	const struct of_device_id *match;
	struct imx_pinctrl_soc_info *pinctrl_info;
    /*
    const struct of_device_id *of_match_device(const struct of_device_id *match,
                                              const struct device *dev);
    Description
    Used by a driver to check whether an platform_device present in the system is in its list of supported devices.
    */
	match = of_match_device(imx6ul_pinctrl_of_match, &pdev->dev);

	if (!match)
		return -ENODEV;

	pinctrl_info = (struct imx_pinctrl_soc_info *) match->data;
	return imx_pinctrl_probe(pdev, pinctrl_info);
}

官方自己的pin头文件，里定义了一组宏，表示一组寄存器的定义，最终通过解析这个宏来注册引脚的寄存器配置

struct imx_pinctrl_soc_info {
	struct device *dev;
	const struct pinctrl_pin_desc *pins;

观察官方的dts会发现除了官方自己的宏，后边还跟了一个变量，这个变量是config，对应conf_reg。通过自定义config我们配置对应引脚的寄存器模式。

编写myled.c#

static int led_probe(struct platform_device *pdev) {
	int gpio, ret;
	gpio = of_get_named_gpio(pdev->dev.of_node, "led-gpios", 0);
	if(gpio < 0) {
		printk("get led-gpio failed: %d\n", gpio);
		return gpio;
	}

	ret = gpio_request(gpio, "myled");
	if (ret) {
		printk("gpio request failed\n");
		return ret;
	}

	ret = gpio_direction_output(gpio, 1); // 板子上电启动后灯是亮的，因此在这里把它关了
	if(ret) {
		printk("gpio direction failed\n");
		gpio_free(gpio);
		return ret;
	}

	return 0;
}

static int led_remove(struct platform_device *pdev) {
	printk("led remove\n");
	return 0;
}

platform_driver#

static const struct of_device_id led_of_match[] = {
	{.compatible = "led-gpio"},
	{},
};

static struct platform_driver led_driver = {
	.probe = led_probe,
	.remove = led_remove,
	.driver = {
		.name = "led",
		.of_match_table = led_of_match,
	},
};

register and unregister#

这一块是一个固定模版。当内核加载和卸载驱动程序时，会调用__init和__exit函数。

最简单的驱动程序就是在这两个函数中加printk来打印信息。在加载和卸载时就会在终端上打印。

常用命令：

  • insmod 加载
  • rmmod 卸载
  • lsmod 列出加载到内核的所有模块
  • depmod 更新模块间的依赖关系
  • modinfo 模块的信息-> filename/license/description/depends/vermagic/parm
  • modprobe 加载时自动发现依赖关系
static int __init led_init(void) {
	return platform_driver_register(&led_driver);
}

static void __exit led_exit(void) {
	platform_driver_unregister(&led_driver);
}

module_init(led_init);
module_exit(led_exit);

#include <linux/kernel.h>
#include <linux/device.h>
#include <linux/gpio/consumer.h>
#include <linux/platform_device.h>
#include <asm/uaccess.h>

struct led_dev {
	struct cdev cdev;
	dev_t devid;
	struct class *class;
	struct device *device;
	struct gpio_desc *led_gpio;
};

static int led_open(struct inode *inode, struct file *file) {
	struct led_dev *led = container_of(inode->i_cdev, struct led_dev, cdev);

	file->private_data = led;

	return 0;
}

static ssize_t led_write(struct file *file, const char __user *user_buffer, size_t size, loff_t * offset) {
	struct led_dev *led = file->private_data;
	char kernel_buffer;

	if (size < 1)
		return -EINVAL;
	if (copy_from_user(&kernel_buffer, user_buffer, 1))
		return -EINVAL;

	if (kernel_buffer == '1')
		gpiod_set_value(led->led_gpio, 1);
	else if ( kernel_buffer == '0')
		gpiod_set_value(led->led_gpio, 0);
	else
		return -EINVAL;

	return size;
}

static struct file_operations led_fops = {
	.owner = THIS_MODULE,
	.open = led_open,
	.read = led_read,
	.write = led_write,
	.release = led_release,
};

static int led_probe(struct platform_device *pdev) {
	struct led_dev *led;
	int ret;

	led = devm_kzalloc(&pdev->dev, sizeof(*led), GFP_KERNEL);
	if (!led)
		return -ENOMEM;
	//led->dev = &pdev->dev;

	platform_set_drvdata(pdev, led);

	led->led_gpio = devm_gpiod_get(&pdev->dev, "led", GPIOD_OUT_LOW);
	if (IS_ERR(led->led_gpio)) {
		ret = PTR_ERR(led->led_gpio);
		dev_info(&pdev->dev, "failed to get led gpio: %d\n", ret);
		return ret;
	}

	ret = alloc_chrdev_region(&led->devid, 0, 1, "led");
	if (ret)
		return ret;

	cdev_init(&led->cdev, &led_fops);
	led->cdev.owner = THIS_MODULE;
	ret = cdev_add(&led->cdev, led->devid, 1);
	if (ret)
		goto error_unregister_chrdev;

	led->class = class_create(THIS_MODULE, "led_class");
	if (IS_ERR(led->class)) {
		ret = PTR_ERR(led->class);
		goto error_cdev_del;
	}

	led->device = device_create(led->class, NULL, led->devid, NULL, "led1");
	if (IS_ERR(led->device)) {
		ret = PTR_ERR(led->device);
		goto error_class_destroy;
	}
	dev_info(&pdev->dev, "led probe\n");
	return 0;

error_class_destroy:
	class_destroy(led->class);

error_cdev_del:
	cdev_del(&led->cdev);

error_unregister_chrdev:
	unregister_chrdev_region(led->devid, 1);

	return ret;
}

static int led_remove(struct platform_device *pdev) {
	struct led_dev *led;

	led = platform_get_drvdata(pdev);

	device_destroy(led->class, led->devid);
	class_destroy(led->class);
	cdev_del(&led->cdev);
	unregister_chrdev_region(led->devid, 1);

	dev_info(&pdev->dev, "led removed\n");

	return 0;
}

错误1#

错误2#

devm_gpiod_get(&pdev->dev, “led-gpios”, GPIOD_OUT_LOW);

这里的字符串匹配机制跟of的不一样

  • of是led-gpios->led-gpios
  • devm是led->led-gpios xxx->xxx-gpios
