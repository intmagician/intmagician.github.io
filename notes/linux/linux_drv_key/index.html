非中断版本#

跟led驱动大体结构一致，多写一个read函数，用gpiod_get_value

static ssize_t key_read(struct file *file, char __user *user_buffer, size_t size, loff_t *offset)
{
    struct key_dev *key = file->private_data;
    int value;
    if (*offset > 0)
        return 0;

    value = gpiod_get_value(key->key_gpio);

    if(copy_to_user(user_buffer, &value, sizeof(value)))
        return -EFAULT;

    return sizeof(value);
}

input子系统#

误打误撞，用了gpio-keys的框架，驱动啥也不用干，dts按照dts-binding写好，启动内核时就能找到

input: gpio-keys as /devices/platform/gpio-keys/input/input2

费这么大劲写input驱动，毛用没有，问ChatGPT，才知道，gpio-keys是官方写好驱动了，就加个gpio-keys node，然后列出一堆key，系统在启动时会注册gpio-keys。

struct key_dev
{
    struct input_dev *input;
    struct gpio_desc *gpiod;

    int irq;
    struct delayed_work dwork;
}

//删除字符设备注册那一个流程

static int key_probe(struct platform_device *pdev)
{
+   key->input = devm_input_allocate_device(&pdev->dev);
+   key->input->name = "key";
+   input_set_capability(key->input, EV_KEY, KEY_A);
+   ret = input_register_device(key->input);
+   if (ret)
+   {
+       dev_info(&pdev->dev, "input register failed: %d\n", ret);
+       return ret;
+   }
}

static int key_remove(struct platform_device *pdev)
{
    struct key_dev *key = platform_get_drvdata(pdev);

    cancel_delayed_work_sync(&key->dwork);

    dev_info(&pdev->dev, "key removed\n");
    return 0;
}

static irqreturn_t key_irq_handler(int irq, void *dev_id)
{
    struct key_dev *key = dev_id;
+    mod_delayed_work(system_wq, &key->dwork, jiffies + msecs_to_jiffies(20));
}

static void key_work_func(struct work_struct *work)
{
    struct key_dev *key = container_of(work, struct key_dev, dwork.work);
    int value;

    value = gpiod_get_value(key->gpiod);
    input_report_key(key->input, KEY_A, value);
    input_sync(key->input);
    //
}
