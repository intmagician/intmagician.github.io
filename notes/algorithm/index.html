NC4: 判断链表是否有环#

知识点：

  • 链表
  • 快慢指针

主要思路：一个指针跑得快，一个指针跑得慢，如果有环，它俩一定会相遇。

struct ListNode * fast = head;
struct ListNode * slow = head;

while(fast != NULL && fast->next != NULL) {
    fast = fast->next->next;
    slow = slow->next;
    if(fast == slow)
        return true;
}

队列#

First In First Out 实现：

type queue[size];
int front,rear;

enqueue();
dequeue();
is_empty();
is_full();

//循环队列/环形缓冲
empty: front == rear;
full: (rear + 1) % size == front;
rear = (rear + 1) % size;
front = (front + 1) % size;

应用场景：

  • UART接收数据
  • CAN消息收发
  • RTOS消息队列
  • 循环队列/环形缓冲

queue[rear++] = root;

    struct TreeNode * node = queue[front++];

    //

        queue[rear++] = node->left;
    }

        queue[rear++] = node->right;

NC90:包含main函数的栈

struct stack_t {
    int array[300];
    int top;
    int ele_num;
};
struct stack_t stack = {
    .top = 0,
    .ele_num = 0,
};


void push(int value) {
    stack.array[stack.top++]= value;
    stack.top++;
    stack.ele_num++;
}

void pop() {
    if (stack.ele_num > 0) {
        stack.top--;
        stack.ele_num--;
    }
}

int top() {
    return stack.array[stack.top - 1];
}

int min() {
    if(stack.ele_num == 0)
        return 0;
    int min = stack.array[stack.top - 1];
    int tmp = stack.top - 1;
    for (int i = 0; i < stack.ele_num; i++) {
        if (min > stack.array[tmp]) {
            min = stack.array[tmp];
        }
        tmp--;
    }
    return min;
}

int left = 1;
int right = x / 2;

while (left <= right) {
    int mid = left + (right - left) / 2;

    if (mid <= x / mid) {
        left = mid + 1;
        //取值向下取整数
    }
    else {
        right = mid - 1;

    }
